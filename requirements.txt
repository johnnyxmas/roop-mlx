# ROOP-FLOYD-MLX -- Apple Silicon only.
# This fork drops the ONNX Runtime / CUDA / TensorRT / DirectML / ROCm
# stack from the upstream repo and targets MLX for all model inference.

# -- MLX --
mlx>=0.20.0; sys_platform == "darwin" and platform_machine == "arm64"

# -- onnxruntime for the CoreML execution provider: 2-4x faster than the
# MLX interpreter for the detector stack and large enhancers (see
# roop/inference.py for the per-model backend split).
# 1.28 is the floor: 1.27 ships no cp314 wheel, so an older bound silently
# blocks python 3.14.  CoreML EP and swap parity verified on 1.28.
onnxruntime>=1.28,<2.0; sys_platform == "darwin" and platform_machine == "arm64"

# -- diffusion tab (roop/diffusion): the SD 1.5 / SDXL / FLUX stacks are
# implemented directly in MLX, so no torch or diffusers is pulled in.
# huggingface-hub fetches checkpoints on demand; tokenizers reads the T5
# vocabulary FLUX conditions on (CLIP reuses the vendored BPE in clip/).
huggingface-hub>=0.30,<2.0
tokenizers>=0.20,<1.0

# -- core numerics / image / video --
# opencv 5 (2026-07): verified against every cv2 symbol this repo calls (59 of

# filters) plus a real end-to-end swap -- output differs from 4.12 only by
# interpolation rounding in the blended face region (max 39/765, p50 1).
# 5.x also drops opencv's numpy<2.3 ceiling, which is why numpy is unbounded
# again here.

opencv-python-headless>=5.0,<6.0
scipy>=1.11,<2.0
scikit-image>=0.21,<1.0
